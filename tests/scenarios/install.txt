# The Arch-style install: partition, format, mount, brickstrap, reboot into it.
alice: fdisk -l /dev/vda
@expect alice Disk /dev/vda
alice: !keys fdisk /dev/vda\r
alice: !keys o\r
alice: !keys n\r
alice: !keys p\r
alice: !keys 1\r
alice: !keys \r
alice: !keys \r
alice: !keys w\r
@wait 3
alice: cat /proc/partitions
@expect alice vda1
alice: mkfs.ext2 -q /dev/vda1
alice: mount /dev/vda1 /mnt && df -h /mnt
@expect alice /mnt
alice: brickstrap /mnt cowsay
@expect alice Done.
alice: echo mybox > /mnt/etc/hostname; sync; umount /mnt; echo UNMOUNTED
@expect alice UNMOUNTED
@reboot alice
@wait 15
@expect alice mybox login:
alice: !keys root\r
@wait 5
alice: hostname; cat /etc/os-release | head -1; brick list
@expect alice cowsay 1.0
alice: brick install fortune
alice: fortune | cowsay; echo persistent-file > /root/note.txt; sync
@expect alice (oo)
@reboot alice
@wait 15
alice: !keys root\r
@wait 5
alice: cat /root/note.txt
@expect alice persistent-file
